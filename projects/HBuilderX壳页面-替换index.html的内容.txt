    var APP_URL = 'https://xiaoluo-xiaoluo.pages.dev';

    // 监听 5+ 引擎就绪，创建一个全屏网页窗口加载网站
    document.addEventListener('plusready', function () {
      var ws = plus.webview.create(APP_URL, 'main', {
        top: '0px',
        bottom: '0px',
        statusbar: { background: '#F7F2E6' }
      });
      // 网站加载完成后撤掉启动画面
      ws.addEventListener('loaded', function () {
        var s = document.getElementById('splash');
        s.style.opacity = '0';
        setTimeout(function () { s.style.display = 'none'; }, 500);
      });
      plus.webview.show(ws);
    }, false);
  </script>
